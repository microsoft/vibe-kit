# Dayhoff Atlas Data Integration

## Supported Formats

**FASTA**: Unaligned protein sequences for unconditional generation prompts or benchmarking (source: Dayhoff Atlas dataset card).  
**Arrow/Parquet**: Columnar subsets for large-scale sampling or fine-tuning pipelines (source: Dayhoff Atlas dataset card).  
**JSONL**: Metadata-rich records with provenance tags for analytics and filtering (source: Dayhoff Atlas dataset card).

## Loading Examples

```python
"""Stream metagenomic representatives from the GigaRef cluster FASTA."""
from datasets import load_dataset

dataset = load_dataset(
    "microsoft/Dayhoff", name="gigaref_no_singletons", split="train", streaming=True
)

def iterate_sequences(limit=5):
    for idx, row in enumerate(dataset):
        yield row["sequence"]
        if idx + 1 == limit:
            break

sample_sequences = list(iterate_sequences())
assert all(set(seq.upper()) <= set("ACDEFGHIKLMNPQRSTVWY") for seq in sample_sequences)
```

```python
"""Load BackboneRef novelty sequences from Arrow into a pandas DataFrame."""
from datasets import load_dataset

backbone = load_dataset(
    "microsoft/Dayhoff",
    name="backboneref",
    split="BBR_n",
    streaming=False,
)

df = backbone.to_pandas()[["sequence", "backbone_id", "predicted_tm"]]
assert df["sequence"].str.len().between(30, 512).all()
```

```python
"""Filter DayhoffRef JSONL for sequences generated by Dayhoff-3b-GR-HM-c."""
import json
from datasets import load_dataset

dayhoff_ref = load_dataset("microsoft/Dayhoff", name="dayhoffref", split="train", streaming=True)

selected = []
for row in dayhoff_ref:
    meta = json.loads(row["metadata"])
    if meta.get("model_name") == "Dayhoff-3b-GR-HM-c":
        selected.append({"sequence": row["sequence"], "meta": meta})
    if len(selected) == 10:
        break

assert len(selected) == 10
```

## Validation

```python
def validate_dayhoff_batch(records):
    return all(
        10 <= len(rec["sequence"]) <= 2000 and set(rec["sequence"]) <= set("ACDEFGHIKLMNPQRSTVWY")
        for rec in records
    )

assert validate_dayhoff_batch(selected)
print("Dayhoff data integration checks passed.")
```
